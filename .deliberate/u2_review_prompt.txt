+/// Orthonormal disc basis perpendicular to a (unit) pour direction `dir`.
+fn disc_basis(dir: [f32; 3]) -> ([f32; 3], [f32; 3]) {
+    let cross = |a: [f32; 3], b: [f32; 3]| {
+        [
+            a[1] * b[2] - a[2] * b[1],
+            a[2] * b[0] - a[0] * b[2],
+            a[0] * b[1] - a[1] * b[0],
+        ]
+    };
+    let norm = |v: [f32; 3]| {
+        let l = (v[0] * v[0] + v[1] * v[1] + v[2] * v[2]).sqrt().max(1.0e-9);
+        [v[0] / l, v[1] / l, v[2] / l]
+    };
+    // Reference axis not parallel to dir.
+    let refv = if dir[1].abs() < 0.9 {
+        [0.0, 1.0, 0.0]
+    } else {
+        [1.0, 0.0, 0.0]
+    };
+    let u = norm(cross(dir, refv));
+    let w = cross(dir, u);
+    (u, w)
+}

+        solver.active_count() > 50,
+        "too little emitted ({}) to be a meaningful stability test",
+        solver.active_count()
+    );
+    assert!(
+        pos.iter()
+            .all(|p| p[0].is_finite() && p[1].is_finite() && p[2].is_finite()),
+        "non-finite position (inlet eruption)"
+    );
+    let vmax = vel
+        .iter()
+        .map(|v| (v[0] * v[0] + v[1] * v[1] + v[2] * v[2]).sqrt())
+        .fold(0.0f32, f32::max);
+    assert!(
+        vmax <= cfg.max_speed + 1.0,
+        "velocity blew past the cap (eruption): vmax {vmax}"
+    );
+}
